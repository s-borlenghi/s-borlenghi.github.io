This is how I'd serve this model with FastAPI. Try a request: the answer comes from the network training in your browser, not from a server.

from fastapi import FastAPI
from pydantic import BaseModel, Field
import torch

app = FastAPI()
model = torch.jit.load("model.pt").eval()

class Point(BaseModel):
    x: float = Field(ge=-1, le=1)
    y: float = Field(ge=-1, le=1)

@app.post("/predict")
def predict(p: Point):
    with torch.no_grad():
        prob = model(torch.tensor([[p.x, p.y]])).item()
    return {"class": int(prob > 0.5), "probability": round(prob, 3)}
