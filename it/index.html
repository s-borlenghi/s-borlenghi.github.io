Così servirei questo modello con FastAPI. Prova una richiesta: la risposta arriva dalla rete che si sta addestrando nel tuo browser, non da un server.

from fastapi import FastAPI
from pydantic import BaseModel, Field
import torch

app = FastAPI()
model = torch.jit.load("model.pt").eval()

class Point(BaseModel):
    x: float = Field(ge=-1, le=1)
    y: float = Field(ge=-1, le=1)

@app.post("/predict")
def predict(p: Point):
    with torch.no_grad():
        prob = model(torch.tensor([[p.x, p.y]])).item()
    return {"class": int(prob > 0.5), "probability": round(prob, 3)}

Un esempio di analisi con Pandas. Il pulsante analizza davvero le previsioni della rete che si sta addestrando nel tuo browser.

import pandas as pd

df = pd.DataFrame(punti, columns=["x", "y", "classe"])
prob = modello.predict(df[["x", "y"]])
df["previsione"] = (prob > 0.5).astype(int)

# Matrice di confusione e accuratezza
print(pd.crosstab(df["classe"], df["previsione"]))
print(f"Accuratezza: {(df.classe == df.previsione).mean():.1%}")
